class GraphState(TypedDict):

    Represents the state of our graph.

    Attributes:

        messages: Conversation history

def identity_complete_status(row):
    if row.get('passport_col') is not None or row.get('ssn_col') is not None:
        return "Complete"
    else:
        return "Incomplete"

def address_complete_status(row):
    if row.get('utility_bill_col') is not None:
        return "Complete"
    else:
        return "Incomplete"

def overall_complete_status(row):
    if row.get('Address_complete_status') == "Complete" and row.get('Identity_complete_status') == "Complete":
        return "Complete"
    else:
        return "Incomplete"

def mongodb_connect(customer_name: str) -> str:
    """
    Connects to MongoDB, retrieves document filenames for a specific customer.
    Expects customer_name as input. Returns a string summary with a parsable format.
    """
    mongodb_client_url = "mongodb://localhost:27017/"
    mongodb_database_name = "KYC_Customer_Docs"
    collection_name = customer_name

    try:
        client = MongoClient(mongodb_client_url)
        client.admin.command('ismaster')
        db = client[mongodb_database_name]

        if collection_name not in db.list_collection_names():
             client.close()
             return f"STATUS: ERROR, CUSTOMER: {customer_name}, MESSAGE: Could not find the collection '{collection_name}' for database '{mongodb_database_name}'. Please check the customer name."

        collection = db[collection_name]
        filenames = collection.distinct("filename")
        client.close()

        if filenames:
             return f"STATUS: SUCCESS, CUSTOMER: {customer_name}, FILENAMES: {', '.join(filenames)}"
        else:
             return f"STATUS: SUCCESS, CUSTOMER: {customer_name}, FILENAMES: "

    except Exception as e:
        return f"STATUS: ERROR, CUSTOMER: {customer_name}, MESSAGE: An error occurred: {e}"


# --- Tool Function 2: Check Document Completeness (Modified Output) ---
def check_docs_completeness(mongodb_output_string: str) -> str:
    """
    Processes the output string from the mongodb_connect_tool to check document completeness.
    Expects a string like 'STATUS: SUCCESS, CUSTOMER: ..., FILENAMES: ...'
    Returns a string indicating the completeness status including Address and Identity details.
    """
    #print("\\\\--------------------Calling Document Completeness Check Tool--------------------\\\\")

    status_match = re.search(r"STATUS: (SUCCESS|ERROR)", mongodb_output_string)
    customer_match = re.search(r"CUSTOMER: (.*?)(?:,|$)", mongodb_output_string)
    filenames_match = re.search(r"FILENAMES: (.*)", mongodb_output_string)
    message_match = re.search(r"MESSAGE: (.*)", mongodb_output_string)

    status = status_match.group(1) if status_match else None
    customer_name = customer_match.group(1).strip() if customer_match else "Unknown Customer"
    filenames_str = filenames_match.group(1).strip() if filenames_match else ""
    error_message = message_match.group(1).strip() if message_match else None

    if status == "ERROR":
        #print(f"Received error status from previous tool: {error_message}")
        return f"Previous step failed: {error_message}. Cannot check document completeness."

    if status != "SUCCESS":
         #print(f"Received unexpected status from previous tool: {mongodb_output_string}")
         return f"Could not parse expected status from previous tool output. Cannot check completeness."

    filenames = [name.strip() for name in filenames_str.split(',') if name.strip()]

    # print(f"Checking completeness for customer: {customer_name}")
    # print(f"Documents found: {filenames if filenames else 'None'}")

    utility_bill_exists = any("bill" in doc.lower() for doc in filenames)
    passport_exists = any("passport" in doc.lower() for doc in filenames)
    ssn_exists = any("ssn" in doc.lower() for doc in filenames)
    # kyc_form_exists = any("kyc application form" in doc.lower() for doc in filenames) # Not used in logic below

    doc_status = {
        'utility_bill_col': "Found" if utility_bill_exists else None,
        'passport_col': "Found" if passport_exists else None,
        'ssn_col': "Found" if ssn_exists else None,
    }

    identity_status = identity_complete_status(doc_status)
    address_status = address_complete_status(doc_status)
    overall_status = "Complete" if identity_status == "Complete" and address_status == "Complete" else "Incomplete"

    # print(f"Address Status: {address_status}, Identity Status: {identity_status}, Overall Status: {overall_status}")
    # print(f"------------------Document Completeness check completed --------------------\n\n\n")

    # --- Return a string indicating status and details ---
    # MODIFIED: Include Address and Identity statuses in the return string
    summary = f"Completeness Status for '{customer_name}': Overall: {overall_status}, Address: {address_status}, Identity: {identity_status}. "

    if overall_status == "Incomplete":
        missing_docs_list = []
        if address_status == "Incomplete":
            missing_docs_list.append("Address Proof (Utility Bill)")
        if identity_status == "Incomplete":
             if not passport_exists and not ssn_exists:
                missing_docs_list.append("Identity Proof (Passport or SSN)")
             elif not passport_exists:
                 missing_docs_list.append("Identity Proof (Passport is missing)")
             elif not ssn_exists:
                 missing_docs_list.append("Identity Proof (SSN is missing)")

        summary += "Missing documents: " + ", ".join(missing_docs_list) if missing_docs_list else "Details unavailable (check statuses)."

    return summary

def remove_ansi_codes(text):
    """
    Removes ANSI escape codes from a string.
    """
    # This is a common regex pattern to match ANSI escape codes
    ansi_escape = re.compile(r'\x1B(?:[@-Z\\-_]|\[[0-9;]*[0-9A-Za-z])')
    return ansi_escape.sub('', text)


import re

def format_agent_trace(text):
    """
    Formats the cleaned agent trace by ensuring key markers are on new lines.
    Assumes the input text has already had ANSI codes removed.
    """
    # Add a newline before specific markers, but only if not already preceded by a newline or start of string
    # Using lookbehind (?<!...) to avoid adding multiple newlines if one exists
    # Added check for start of string using |^
    text = re.sub(r'(?<![\n>])Action:', r'\nAction:', text)
    text = re.sub(r'(?<!\n)Action Input:', r'\nAction Input:', text)
    text = re.sub(r'(?<!\n)Final Answer:', r'\nFinal Answer:', text)

    # Ensure chain start/end markers are on their own lines for clarity
    text = text.replace("> Entering new AgentExecutor chain...", "> Entering new AgentExecutor chain...\n")
    text = text.replace("> Finished chain.", "\n> Finished chain.")

    # Optional: Clean up potential multiple consecutive newlines that might result
    text = re.sub(r'\n{2,}', r'\n', text)

    # Trim leading/trailing whitespace that might result from added newlines
    text = text.strip()

    return text

def supervisor_agent_node(state: GraphState) -> GraphState:
    """
    Supervisor node that decides the next agent to call or if the process is finished.
    Updates the state with the decision in the 'next' field.
    Includes refined logic for finishing after the email step.
    """
    print("\n--- Executing supervisor_node ---")
    query = state['query']
    mongodb_output = state.get('mongodb_agent_output')
    completeness_summary = state.get('completeness_agent_summary')
    final_response = state.get('final_response') # Get the final response from potentially the Gmail agent
    error = state.get('error')
    messages = state.get('messages', [])

    supervisor_llm_prompt = ChatPromptTemplate.from_messages([
        ("system", """You are a supervisor agent. Your role is to direct the workflow based on the user's query
         and the current state of the process.
         Respond ONLY in the form of a dictionary consists of two keys where first key 'decision' is the name of the next worker or `FINISH` and second key 'Reason' illustrated the reason of the decision you make.

         The available workers are:
         - `DocuSeek`: Use this to retrieve document filenames for a customer from MongoDB. Requires a customer name.
         - `DocuHunt`: Use this to analyze the output of the mongodb_agent and determine the KYC document completeness status. Requires the output string from the mongodb_agent.
         - `MailCast`: Use this to send an email to the customer based on the completeness status. Requires the completeness summary from the completeness_agent.

         The typical process flow for a KYC completeness check is:
         1. Receive query about a customer's KYC completeness.
         2. Call `DocuSeek` to get documents.
         3. Call `DocuHunt` to check completeness based on mongodb output.
         4. If documents are incomplete based on the completeness summary, call `MailCast` to send an email to the customer about missing documents.
         5. If documents are complete (as indicated in the completeness summary), the task is `AI Supervisor AccuSure`.
         6. If `MailCast` was successfully called (indicated by its output in the state), and the email action is complete, the task is finished.
         7. If an error occurred in any step, the task should finish.

         Current state:
         Query: {query}
         MongoDB Output: {mongodb_output}
         Completeness Summary: {completeness_summary}
         Final Response: {final_response}
         Error: {error}

         Based on the user's query and the current state, decide which worker should act next or if the process is finished (`FINISH`).
         If there is an error in the state, respond with `FINISH`.
         If the 'Completeness Summary' indicates "Overall: Complete", respond with `AI Supervisor AccuSure`.
         If the 'Final Response' indicates that the email action was completed, respond with `FINISH`.
         Respond ONLY in the form of a dictionary consists of two keys where first key 'decision' is the name of the next worker or `FINISH` and second key 'Reason' illustrated the reason of the decision you make.
         Available workers: DocuSeek, DocuHunt, MailCast, FINISH, `AI Supervisor AccuSure`
         """),
        MessagesPlaceholder(variable_name="messages") # Include message history for context
    ])
    st.divider()

    supervisor_input = {
        "query": query,
        "mongodb_output": mongodb_output if mongodb_output is not None else "N/A",
        "completeness_summary": completeness_summary if completeness_summary is not None else "N/A",
        "final_response": final_response if final_response is not None else "N/A",
        "error": error if error is not None else "N/A",
        "messages": messages # Pass messages
    }

    try:
        # Include a summary of the current state in the messages for the supervisor LLM
        state_summary_message = f"""Current State Summary:
                                    Query: {query}
                                    MongoDB Output: {mongodb_output if mongodb_output is not None else "N/A"}
                                    Completeness Summary: {completeness_summary if completeness_summary is not None else "N/A"}
                                    Final Response: {final_response if final_response is not None else "N/A"}
                                    Error: {error if error is not None else "N/A"}
                                """
        # Append the state summary to the messages
        supervisor_input["messages"] = messages + [HumanMessage(content=state_summary_message, name="system_state")]


        decision_output = llm.invoke(supervisor_llm_prompt.format_prompt(**supervisor_input)).content.strip()
        #print(f"Supervisor decided: {decision}")
        #st.text(decision_output)


        #decision_final_output=(decision_output.content)  
        decision_final_output= decision_output.replace("```json","")
        decision_final_output= decision_final_output.replace("```","")
        decision_final_output = ast.literal_eval(decision_final_output)

        decision= decision_final_output.get("decision")
        reason= decision_final_output.get("Reason")
        #st.success(f"Supervisor Accuracy decided: {decision}")
        #st.badge(f"Supervisor Accuracy decided: {decision}")


        # Validate the decision
        valid_decisions = ["DocuSeek", "DocuHunt", "MailCast", "FINISH","AI Supervisor AccuSure"]
        if decision not in valid_decisions:
             st.text(f"Warning: Supervisor returned invalid decision '{decision}'. Defaulting to FINISH.")
             decision = "FINISH"

        # --- Explicit Checks for Finishing Conditions ---
        # This adds robustness in case the LLM doesn't perfectly follow the prompt
        if error:
             #st.text("Error detected in state, forcing FINISH.")
             decision = "FINISH"
        elif completeness_summary and "Overall: Complete" in completeness_summary:
             #st.text("Completeness is complete, forcing Accuracy.")
             decision = "AI Supervisor AccuSure"
             
        # Add a check for a specific phrase in the final_response indicating email completion
        # You might need to adjust the string "Email sent created successfully"
        elif final_response and "Email sent created successfully" in final_response:
             #st.text("Email action completed, forcing FINISH.")
             decision = "FINISH"
        # --- End Explicit Checks ---


        #st.write(f"Supervisor decided: {decision}")
        #st.markdown(":blue-badge[Home]")
        st.badge(f"AI Supervisor DocuSure decided: {decision}",icon=":material/supervisor_account:")
        st.warning(f"Reason for decision: {reason}",icon=":material/supervisor_account:")
        # if decision == "Accuracy":
        #     st.subheader("Document Accuracy")
            
        
        # Update the state with the decision for the conditional edge to read
        updated_state = {**state,
                         "next": decision}
        return updated_state

    except Exception as e:
        print(f"Error in supervisor_agent_node: {e}")
        st.text(e)
        # If supervisor fails, update state to finish
        return {**state, "next": "FINISH",
                'error': f"Error in supervisor decision: {e}",
                'final_response': f"An error occurred while determining the next step: {e}"}

import ast
import re

def parse_ocr_output(output: str, expected_keys: list) -> dict:
    try:
        # Try parsing as Python dictionary first
        return ast.literal_eval(output)
    except:
        # Fallback: parse manually using regex
        result = {}
        for key in expected_keys:
            match = re.search(f"{key}[:=] ?([A-Za-z0-9 \-]+)", output, re.IGNORECASE)
            result[key] = match.group(1).strip() if match else ""
        return result

def llm_similarity_check(text1: str, text2: str):
    """
    Uses LLM to compare two text strings and determine if they are semantically similar.
    Returns a tuple (is_similar: bool, confidence_score: float)
    """
    from langchain_core.output_parsers import StrOutputParser
    from langchain_core.prompts import ChatPromptTemplate

    prompt = ChatPromptTemplate.from_template("""
    Compare the following two texts and determine if they refer to the same entity.

    Text 1: "{text1}"
    Text 2: "{text2}"

    Respond with only a float score between 0.0 and 1.0 indicating similarity (1.0 = exact match, 0.0 = completely different).
    """)

    chain = prompt | llm_accuracy | StrOutputParser()

    response = chain.invoke({"text1": text1, "text2": text2})
    
    score = float(response.strip())
    return round(score, 2)
